# Stage a build-only source copy, retaining the original repo and gameplay files.
# Voice-chat middleware and precompiled-library link directives are removed;
# the standalone practice loadout is customized separately below.
set(S ${CMAKE_BINARY_DIR}/source)
file(COPY ${SOURCE_ROOT}/ DESTINATION ${S})
file(READ ${S}/CSCommon/Include/Config.h config)
string(REPLACE "#define VOICECHAT" "// Baseline is silent: no legacy voice middleware.\n//#define VOICECHAT" config "${config}")
file(WRITE ${S}/CSCommon/Include/Config.h "${config}")
file(GLOB_RECURSE headers ${S}/*.h ${S}/*.hpp ${S}/*.cpp)
foreach(path IN LISTS headers)
    file(READ ${path} content)
    string(REGEX REPLACE "#[ \t]*pragma comment\\(lib,[^\n]*(bullet|ogg_static|vorbis|realspace2|cml/lib)[^\n]*" "// Build links source targets, not archived vendor libs." content "${content}")
    file(WRITE ${path} "${content}")
endforeach()
execute_process(COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/competitive/patch_source.py --source ${S}
    COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/competitive/data.py
    --data /work/reference-data --header ${CMAKE_BINARY_DIR}/competitive_config.hpp COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/competitive/hit_art.py
    --data /work/reference-data --header ${CMAKE_BINARY_DIR}/competitive_hit_art.hpp
    COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/competitive/reward_art.py
    --data /work/reference-data --header ${CMAKE_BINARY_DIR}/competitive_reward_art.hpp
    COMMAND_ERROR_IS_FATAL ANY)

function(module name directory)
    file(GLOB sources ${S}/${directory}/*.cpp ${S}/${directory}/*.c)
    add_library(${name} STATIC ${sources})
    target_include_directories(${name} PRIVATE ${S}/${directory} ${S}/${name})
endfunction()
