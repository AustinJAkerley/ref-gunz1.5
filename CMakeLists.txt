cmake_minimum_required(VERSION 3.24)
project(GunZNative VERSION 0.1.0 LANGUAGES C CXX)

# Only this explicit native graph is built. Never glob historical SDK binaries.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GUNZ_BUILD_CLIENT "Build the native graphical test client" ON)
option(GUNZ_SANITIZE "Enable address/undefined-behavior sanitizers" OFF)
include(CTest)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(enet
    URL https://codeload.github.com/lsalzman/enet/tar.gz/refs/tags/v1.3.18
    URL_HASH SHA256=28603c895f9ed24a846478180ee72c7376b39b4bb1287b73877e5eae7d96b0dd)
FetchContent_MakeAvailable(enet)
target_include_directories(enet PUBLIC ${enet_SOURCE_DIR}/include)
if(WIN32)
    target_link_libraries(enet ws2_32 winmm)
endif()

add_library(gunz_core native/src/world.cpp native/src/map.cpp native/src/protocol.cpp cml/source/MCRC32.cpp)
target_include_directories(gunz_core PUBLIC native/src cml/Include)
target_compile_definitions(gunz_core PUBLIC GUNZ_NATIVE)
if(MSVC)
    target_compile_options(gunz_core PRIVATE /W4)
else()
    target_compile_options(gunz_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(GUNZ_SANITIZE AND NOT MSVC)
    target_compile_options(gunz_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(gunz_core PUBLIC -fsanitize=address,undefined)
endif()
add_library(gunz_net native/src/network.cpp)
target_link_libraries(gunz_net PUBLIC gunz_core enet)
add_executable(gunz-server native/src/server.cpp)
target_link_libraries(gunz-server PRIVATE gunz_net)
add_executable(gunz-probe native/src/probe.cpp)
target_link_libraries(gunz-probe PRIVATE gunz_net)

if(GUNZ_BUILD_CLIENT)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    # This slice is silent. Do not compile unused audio decoders/middleware.
    set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
    set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
    set(USE_AUDIO OFF CACHE BOOL "" FORCE)
    set(SUPPORT_CLIPBOARD_IMAGE OFF CACHE BOOL "" FORCE)
    # SDL supports Wayland inside Flatpak; Desktop uses source-built GLFW elsewhere.
    if(UNIX AND NOT APPLE)
        set(PLATFORM SDL CACHE STRING "raylib window backend")
    endif()
    FetchContent_Declare(raylib
        URL https://codeload.github.com/raysan5/raylib/tar.gz/refs/tags/5.5
        URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e)
    FetchContent_MakeAvailable(raylib)
    # raylib 5.5's custom-config parser includes both sides of this #if.
    # Keep the GPU-skinning value consistent instead of defining it twice.
    get_target_property(raylib_definitions raylib COMPILE_DEFINITIONS)
    list(FILTER raylib_definitions EXCLUDE REGEX "^MAX_MESH_VERTEX_BUFFERS=")
    set_property(TARGET raylib PROPERTY COMPILE_DEFINITIONS ${raylib_definitions})
    if(RL_SUPPORT_MESH_GPU_SKINNING)
        target_compile_definitions(raylib PRIVATE MAX_MESH_VERTEX_BUFFERS=9)
    else()
        target_compile_definitions(raylib PRIVATE MAX_MESH_VERTEX_BUFFERS=7)
    endif()
    add_executable(gunz-client native/src/client.cpp native/src/scene.cpp)
    target_link_libraries(gunz-client PRIVATE gunz_net raylib)
endif()
if(BUILD_TESTING)
    add_executable(gunz-tests native/tests/core_tests.cpp)
    target_link_libraries(gunz-tests PRIVATE gunz_core)
    add_test(NAME native-core COMMAND gunz-tests)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME native-network COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_SOURCE_DIR}/native/tests/network_test.py
        $<TARGET_FILE:gunz-server> $<TARGET_FILE:gunz-probe>)
    set_tests_properties(native-network PROPERTIES TIMEOUT 25)
    add_test(NAME asset-formats COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/native/tests/asset_tests.py)
    if(EXISTS "${CMAKE_SOURCE_DIR}/native/assets/converted/mansion.collision")
        add_executable(gunz-map-tests native/tests/map_tests.cpp)
        target_link_libraries(gunz-map-tests PRIVATE gunz_core)
        add_test(NAME mansion-collision COMMAND gunz-map-tests
            ${CMAKE_SOURCE_DIR}/native/assets/converted/mansion.collision)
    endif()
endif()
